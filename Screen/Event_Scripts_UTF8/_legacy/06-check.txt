--------------------------------------------------------
【定时器 tm0 事件】  tim=1000, en=0
        ★ 每秒倒计时，推进 5 个步骤 ★
--------------------------------------------------------
if(va6.val<6)
{
  va7.val-=1
  n_cnt.val=va7.val

  if(va7.val<=0)
  {
    // ===== 当前步骤结束，通知下位机停止采集 =====
    // CMD=0x20(32) LEN=2 DATA=[0x03 步骤结束][步骤号]
    // SUM = (32 + 2 + 3 + 步骤号)
    sys0=(34+3+va6.val)%256
    printh AA 55 20 02 03
    prints va6.val,1
    prints sys0,1

    va6.val+=1

    if(va6.val>5)
    {
      // ===== 5 步全部完成，请求下位机判定 =====
      va6.val=6
      // DATA=[0x04 请求判定][0]
      // SUM = (32 + 2 + 4 + 0) = 38 = 0x26
      printh AA 55 20 02 04 00 26

      t_step.txt="等待结果"
      t_hint1.txt="自检完成"
      t_hint2.txt="等待下位机判定..."
      t_state.txt="请稍候"
      t_state.pco=65504
      j0.val=100
      tm0.en=0
    }else
    {
      // ===== 进入下一步骤，通知下位机开始采集 =====
      // DATA=[0x02 步骤开始][步骤号]
      // SUM = (32 + 2 + 2 + 步骤号)
      sys0=(34+2+va6.val)%256
      printh AA 55 20 02 02
      prints va6.val,1
      prints sys0,1

      va7.val=5
      n_cnt.val=5
      j0.val=va6.val*20

      // 步骤 2：拨 1 弦
      if(va6.val==2)
      {
        t_step.txt="步骤 2 / 5"
        t_hint1.txt="请拨 1 弦"
        t_hint2.txt="5 秒内拨动 3 ~ 4 次"
      }
      // 步骤 3：拨 2 弦
      if(va6.val==3)
      {
        t_step.txt="步骤 3 / 5"
        t_hint1.txt="请拨 2 弦"
        t_hint2.txt="5 秒内自由拨动"
      }
      // 步骤 4：拨 3 弦
      if(va6.val==4)
      {
        t_step.txt="步骤 4 / 5"
        t_hint1.txt="请拨 3 弦"
        t_hint2.txt="5 秒内自由拨动"
      }
      // 步骤 5：拨 4 弦
      if(va6.val==5)
      {
        t_step.txt="步骤 5 / 5"
        t_hint1.txt="请拨 4 弦"
        t_hint2.txt="5 秒内自由拨动"
      }
    }
  }
}

================================================================
  自检帧校验和速查表
================================================================
CMD=0x20(32), LEN=2, 基数 34

  自检开始     AA 55 20 02 01 00 23    (34+1+0=35)
  步骤开始 N   AA 55 20 02 02 0N --    (34+2+N)
    步骤1  AA 55 20 02 02 01 25
    步骤2  AA 55 20 02 02 02 26
    步骤3  AA 55 20 02 02 03 27
    步骤4  AA 55 20 02 02 04 28
    步骤5  AA 55 20 02 02 05 29
  步骤结束 N   AA 55 20 02 03 0N --    (34+3+N)
    步骤1  AA 55 20 02 03 01 26
    步骤5  AA 55 20 02 03 05 2A
  请求判定     AA 55 20 02 04 00 26    (34+4+0=38)
  取消         AA 55 20 02 05 00 27    (34+5+0=39)

================================================================
  ★ 结果灯的实现前提
================================================================
t_r1~t_r4 靠改 bco（背景色）表示结果，不需要图片素材：
  未判定：bco=22979（深木底）+ pco=33808（灰字）
  通过  ：bco=2016 （绿底）+ pco=0    （黑字）
  未过  ：bco=63488（红底）+ pco=0    （黑字）

★ 前提：四个控件的 sta 属性必须是【纯色】，
  否则背景图会盖住 bco，脚本改了颜色也看不见。

拆位用"取余 + 整除"（TJC 无位运算）：
  bit0 → va8.val%2      bit1 → va8.val/2%2
  bit2 → va8.val/4%2    bit3 → va8.val/8%2

================================================================
  排错
================================================================
1. 倒计时不走：检查 tm0 是否 en=1，且 tim=1000。
2. 一直停在"等待下位机判定"：下位机没回传 va5。
   用串口助手看屏有没有发出 AA 55 20 02 04 00 26。
3. 步骤文案不切换：va6 递增逻辑有问题，或 va7 没重置为 5。
4. 结果一直灰色：va5 仍是 1（进行中），下位机没改写它。
5. 自检成功但某弦显示"未过"：
   说明 va8 掩码里该位为 0，下位机判定该弦未通过 —— 这是正常的，
   此时应把 va5 设为 3（失败）。若 va5=2 却有弦未过，检查下位机判定逻辑。
