【后初始化事件】
tm0.en=1
// 唤醒屏幕后请求整机唤醒；重复请求不会重复标定/清空正在演奏的引擎。
vpop=2
vpextra=0
vpowerseq=vpowerseq+1
if(vpowerseq>65535)
{
  vpowerseq=0
}
vpseqlo=vpowerseq%256
vpseqhi=vpowerseq/256
vpcrc=28
vpcrc=vpcrc^5
vpcrc=vpcrc^1
vpcrc=vpcrc^vpop
vpcrc=vpcrc^vpseqlo
vpcrc=vpcrc^vpseqhi
vpcrc=vpcrc^vpextra
printh AA 1C 05 01
prints vpop,1
prints vpseqlo,1
prints vpseqhi,1
prints vpextra,1
prints vpcrc,1
