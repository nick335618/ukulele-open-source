// ============================================================
//  ★ 1. 全局变量声明 —— 必须全部写在最前面
//  ------------------------------------------------------------
//  TJC 有两种变量：
//    int  （本工程用它） → Program.s 里声明，代码里直接写名字，不加 .val
//    va 变量控件          → 代码里必须写 va0.val，内存贵十倍
//
//  全部 int，约 60 个 × 4 字节 = 240 字节。
//  ★ 初值必须等于 FPGA 的上电复位值（gvlink.h 的 GvParam 初值），
//    否则"还没回读过"的那几秒，屏上显示的是错的值。
// ============================================================

// ---------- 光电实时值（ESP32 写，屏只读）----------
int vra0=2500,vra1=2500,vra2=2500,vra3=2500
int vdg0=0,vdg1=0,vdg2=0,vdg3=0
int vev=0
int vlink=0
// ★ 链路心跳（ESP32 写 vbeat，屏只看它变没变）
//   比"比较 ADC 总和"可靠得多：环境稳定的时候读数可能几十拍都不变，
//   会被误判成离线。心跳是下位机每次收到 AA 0B 就 +1，只要它在动就说明活着。
int vbeat=0         // ESP32 每收到一次 AA 0B 就写一个新值
int vbeatLast=255   // 上次看到的心跳值（255 是故意的非法值，保证第一拍会更新）
int vbeatAge=0      // 心跳没变的拍数，超过 15 拍判离线

// ---------- 光电参数镜像（ESP32 回读写入，屏只读）----------
//   pid 0..3   阈值   th0..th3      12 位，默认 97
//   pid 4..7   增益   gain0..3       8 位，默认 68/48/78/71
//   pid 8..16  全局参数               默认见 docs/01 §2
int vp0=97,vp1=97,vp2=97,vp3=97
int vp4=68,vp5=48,vp6=78,vp7=71
int vp8=262144,vp9=8192,vp10=0,vp11=10
int vp12=20,vp13=7000,vp14=90000,vp15=40
int vp16=50000

// ---------- 发帧用（屏自己写）----------
int vseq=0
int vopr=0,voch=0,vval=0
int vlo=0,vmid=0,vhi=0,vcrc=0
int vidx=0
int vstage=0       // 两段式发送：1=写参数待发 2=写完后请求回读

// ---------- 页面本地状态 ----------
int vsel=0
int vthpid=0,vgnpid=4
int vraidx=0,vvevbit=1
int vpset=1
int vcfm=0
int vack=0,vackt=0
int vapid=8,vamax=255,vcur=0      // adv 页：当前项 pid / 上限 / 当前值
int vwt=180,vlsel=9               // gain 页：待写入值 / 上次选中（9=强制首拍刷新）
// ★ 刚写完的那个 pid（0~7）。回读确认必须拿它去比对，
//   不能用 vsel —— vsel 是"选了第几行"，调增益时 pid 是 4~7，两者差 4。
int vcheckpid=0

// ---------- 自检状态机（check 页）----------
int vst=0,vstp=1,vtk=0,vsec=0,vrem=5,vprog=0
int vcnt=0,vneed=3,vbit=1,vcd=0,vprev=0
int vr0=0,vr1=0,vr2=0,vr3=0
int vnoise=0
int vraw=0,vthr=180,vnew=0        // 当前弦的强度 / 阈值 / 边沿
// 真实事件指示固定使用vev，禁止将超阈值当作已发声。
int vusev=1
int vfp=0,vfpo=0                  // 结果指纹 / 上次指纹（变了才刷屏）
int vlcnt=0,vlstp=0               // 上次计数 / 上次步骤（同上）

// ---------- 统一屏幕/FPGA休眠：实际状态只能由ESP32回读写入 ----------
int vpowerstate=0,vpowerresult=0,vpowerack=0,vpowercap=0
int vcountdown=10,vsleepmin=5,vpowerseq=0
int vpop=0,vpextra=0,vpseqlo=0,vpseqhi=0,vpcrc=0
// vpowerstate:0工作,1降亮,2提醒,3正在休眠,4已休眠,5正在唤醒,6故障保护
// vpowerresult与演奏模式vopstatus独立；不再把休眠错误显示为模式失败。
// ---------- 演奏模式与真实操作结果（ESP32写入）----------
int vpmwant=0,vpmode=255,vfpga=0,vcfgready=0
int vopstatus=0,vopack=255,voptype=0,vreqseq=0,vopseq=0,vopage=0,vownreq=0
int vdet=0,vacc=0,vemit=0,voq=0,voverflow=0
int vemitcnt=0,vdetectcnt=0,vacceptcnt=0
// vev 是实际送入音频路径的事件，vdg超阈值不等于发声。
// ---------- 音量 / 音色 ----------
//   vavol / vatimbre 这两个名字是【固件写死的】，必须保留
int vavol=70
int vatimbre=0
int vlast=70
int vmute=0

// ---------- LED 灯阵配置（ESP32 保存并写入，屏幕只负责选择）----------
//   mode: 0关 1全常亮 2全亮呼吸 3行/列渐进 4游龙 5行/列波纹 6拨弦跟随
//   axis: 0=行（CS0..CS17） 1=列（SW8/SW9/SW0/SW1）
int vledmode=0,vledaxis=0,vledspeed=24,vledbri=180,vledflags=0

// ---------- 临时量 ----------
int vt0=0,vt1=0,vt2=0,vt3=0


// ============================================================
//  ★ 2. 上电初始化
// ============================================================

// ---------- 波特率 115200 ----------
// 本项目使用 115200（屏出厂为 9600），两侧必须一致。
//
// 【重要】第一次下载时如果 USART HMI 报"找不到设备 / 下载失败"：
//   1) 先把下面这行注释掉，用默认 9600 下载一次
//   2) 确认基础功能跑通后，再取消注释重新下载
//   3) 同时把 USART HMI 下载界面的波特率改成 115200
//   4) 若屏已失联，用 SD 卡下载一个 9600 的工程恢复
bauds=115200

// ---------- 显示 ----------
// 自动休眠由ESP32统一计时；屏幕自身的两个计时器保持关闭。
// thup 控制「休眠后的触摸唤醒」，不是触摸事件/坐标上报开关。
sleep=0          // 冷启动先唤醒，随后同步整机实际状态
thsp=0           // 不再因长时间没有触摸而休眠
ussp=0           // 不再因长时间没有串口数据而休眠
wup=255          // 保留当前页；ESP32按页名导航，禁止假定唤醒页编号13
thup=1           // 轻触唤醒屏幕，ESP32接收通知后恢复FPGA
dim=100          // 冷启动亮度；待机/唤醒由ESP32协调
sendxy=1         // 上报触摸坐标，真实触摸才重置空闲计时
recmod=0         // 0 = 字符串指令协议（ESP32 用 "vra0=2500 FF FF FF" 写变量）

// ---------- 上电广播 ----------
// 屏发这个给 ESP32，让 ESP32 知道"屏活了、可以开始发了"。
// 00 00 00 FF FF FF 是屏幕清零/分帧标记（screen_link.cpp 认识它）。
// 88 FF FF FF 是 TJC 开机成功标记。
printh 00 00 00 ff ff ff
printh 88 ff ff ff

// ---------- 选页码 ----------
// ★ page 必须是 Program.s 的最后一句 —— page 后面的代码永远不执行
page boot

================================================================
  说明
================================================================

1. ★★ 所有 int 变量一律**不加 .val**：
     正确   vra0=2500
     错误   vra0.val=2500     → 屏会回错误码 1A（变量名无效）
   只有"va 变量控件"才要写 .val，本工程一个都不用。

2. ★ 变量必须声明在 Program.s 里，不要用工具箱的"变量控件"：
   内存差 10 倍以上（int 4 字节 vs 控件 + 缓冲）。
   本项目 11 个页面属于控件偏多的工程，RAM（3584 B）是第一条红线。

3. `vavol` / `vatimbre` 这两个名字**不能改** —— 固件里写死了
   （screen_link.cpp 的 setVolume / setTimbreNameIdx）。

4. 编译后**必看输出窗口的"运行内存占用"**，任一页超过 3584 B 就报错。

5. 若报"变量未定义"，说明 Program.s 的声明没生效 ——
   确认这一段是贴在【前初始化事件】里，且 int 声明在最前面。
